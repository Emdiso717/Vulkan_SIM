cmake_minimum_required(VERSION 3.10.0 FATAL_ERROR)

set(NAME xpbdvsrid3d)
set(SRC_DIR ../../../examples/${NAME})
set(BASE_DIR ../../../base)
set(EXTERNAL_DIR ../../../external)

set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -std=c++20 -DVK_USE_PLATFORM_ANDROID_KHR -DVK_NO_PROTOTYPES")
file(GLOB EXAMPLE_SRC "${SRC_DIR}/*.cpp")
add_library(native-lib SHARED ${EXAMPLE_SRC})
add_library(native-app-glue STATIC ${ANDROID_NDK}/sources/android/native_app_glue/android_native_app_glue.c)
add_subdirectory(../base ${CMAKE_SOURCE_DIR}/../base)
set(CMAKE_SHARED_LINKER_FLAGS "${CMAKE_SHARED_LINKER_FLAGS} -u ANativeActivity_onCreate")
include_directories(${BASE_DIR} ${EXTERNAL_DIR} ${EXTERNAL_DIR}/glm ${EXTERNAL_DIR}/eigen ${EXTERNAL_DIR}/imgui ${EXTERNAL_DIR}/tinygltf ${ANDROID_NDK}/sources/android/native_app_glue)
target_link_libraries(native-lib native-app-glue libbase android log z)
